A = {<x1, 0.0>, <x2, 0.3>, <x3, 1.0>, <x4, 0.5>}
B = {<y1, 0.4>, <y2, 0.0>, <y3, 0.3>, <y4, 1.0>}
C = {<x1, 0.7>, <x2, 1.0>, <x3, 0.0>, <x4, 0.2>}

A(x) ~> B(y)
C(x) ~> A(y)
B(x) ~> C(y)
